# ===================== Urban Asset Generation Examples =====================
# Run method (headless Blender): blender -b -P obj_urban_generate.txt --python-exit-code 1
# Ensure infinigen is on PYTHONPATH before running.
# Each example saves a .blend file to ./infinigen/outputs/obj/

import sys
import math
import bpy

sys.path.append("./infinigen")

# Shared material helper (mirrors generate_urban._make_mat)
def _make_mat(name, color, roughness=0.55, metallic=0.0, emission_strength=0.0):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs["Base Color"].default_value = color
        bsdf.inputs["Roughness"].default_value = roughness
        bsdf.inputs["Metallic"].default_value = metallic
        if emission_strength > 0 and "Emission Strength" in bsdf.inputs:
            if len(color) >= 3:
                ec = tuple(color[:3]) + (1.0,)
                if "Emission Color" in bsdf.inputs:
                    bsdf.inputs["Emission Color"].default_value = ec
            bsdf.inputs["Emission Strength"].default_value = emission_strength
    return mat


def clear_scene():
    bpy.ops.object.select_all(action="SELECT")
    bpy.ops.object.delete()


def save(path):
    bpy.ops.wm.save_as_mainfile(filepath=path)
    print(f"Saved: {path}")


# ===================== Example 1: Traffic Light =====================
# Semantic: "Complete traffic light assembly at northwest intersection corner,
#            facing north-south traffic, all three signal lenses present."

clear_scene()

from infinigen.assets.objects.traffic.traffic_light import TrafficLightFactory
from infinigen.assets.utils.urban_primitives import UrbanAssetRequest

mats = {
    "metal":         _make_mat("metal",         (0.18, 0.18, 0.18, 1), roughness=0.62, metallic=0.2),
    "signal_box":    _make_mat("signal_box",     (0.035, 0.04, 0.035, 1)),
    "red_signal":    _make_mat("red_signal",     (1.0, 0.05, 0.03, 1), emission_strength=1.2),
    "yellow_signal": _make_mat("yellow_signal",  (1.0, 0.75, 0.08, 1), emission_strength=0.5),
    "green_signal":  _make_mat("green_signal",   (0.05, 0.9, 0.18, 1), emission_strength=1.0),
}

factory = TrafficLightFactory(mats)
objs, info = factory.create(UrbanAssetRequest(
    asset_type="traffic-light",
    location=(-6.4, 6.4, 0),
    semantic="traffic-light",
    yaw=0.0,
    params={"id": "nw", "facing_axis": "y"},
))
print(f"Traffic light created: {len(objs)} objects, metadata={info}")
save("./infinigen/outputs/obj/traffic_light_nw.blend")

# ===================== Example 18: Reusing Nature Factories (shared with nature stream) =====================
# Urban trees and groundcover reuse nature factories — these are the SAME as obj_nature_generate.py
# TreeFactory (used by UrbanTreeFactory for street trees):

clear_scene()

from infinigen.assets.objects.trees.generate import TreeFactory, BushFactory

factory = TreeFactory(factory_seed=42, season="summer", fruit_chance=0.2, coarse=False)
tree = factory.spawn_asset(42, loc=(0, 0, 0), rot=(0, 0, 0), distance=10, vis_distance=10)
print(f"TreeFactory (urban reuse): {tree.name}")
save("./infinigen/outputs/obj/urban_reuse_tree.blend")

# GrassTuftFactory (used by UrbanGroundcoverFactory):
clear_scene()

from infinigen.assets.objects.grassland.grass_tuft import GrassTuftFactory
from infinigen.core.util.math import FixedSeed

gf = GrassTuftFactory(seed=1234)
with FixedSeed(1234):
    grass = gf.create_asset()
gf.finalize_assets([grass])
grass.scale *= 2.5  # urban scale
print(f"GrassTuftFactory (urban reuse): {grass.name}")
save("./infinigen/outputs/obj/urban_reuse_grass.blend")

# FlowerPlantFactory (used by UrbanGroundcoverFactory):
clear_scene()

from infinigen.assets.objects.grassland.flowerplant import FlowerPlantFactory

fpf = FlowerPlantFactory(factory_seed=5678, coarse=False)
with FixedSeed(5678):
    flower = fpf.create_asset()
flower.scale *= 0.85
print(f"FlowerPlantFactory (urban reuse): {flower.name}")
save("./infinigen/outputs/obj/urban_reuse_flower.blend")
